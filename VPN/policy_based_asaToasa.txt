ASAv1:
en

conf t
!
! ==== INTERFACES ====
interface g0/0
 nameif outside
 security-level 0
 ip address 11.1.1.1 255.255.255.0
 no shutdown

interface g0/1
 nameif inside
 security-level 100
 ip address 172.16.1.11 255.255.255.0
 no shutdown

! ==== IKEv1 Phase 1 ====
crypto ikev1 policy 10
 authentication pre-share
 encryption aes
 hash sha
 group 5
 lifetime 86400

! ==== Tunnel Group ====
tunnel-group 12.1.1.2 type ipsec-l2l
tunnel-group 12.1.1.2 ipsec-attributes
 ikev1 pre-shared-key cisco

! ==== Define Interesting Traffic (Crypto ACL) ====
access-list TUNNEL extended permit ip host 10.1.1.1 host 10.2.2.2

! ==== NAT Exemption for VPN Traffic ====
object	net in_net
host 172.16.1.1
nat (inside,outside) static 11.1.1.11
! ==== IPsec Phase 2 ====
crypto ipsec ikev1 transform-set TSET esp-aes esp-sha-hmac

! ==== Crypto Map ====
crypto map CMAP 10 match address TUNNEL
crypto map CMAP 10 set peer 12.1.1.2
crypto map CMAP 10 set ikev1 transform-set TSET
crypto map CMAP interface outside

! ==== Enable IKEv1 ====
crypto ikev1 enable outside
!
route outside 0 0 11.1.1.10
route inside 10.1.1.0 255.255.255.0 172.16.1.1
!
!
crypto ikev1 enable outside
!
access-list OUTSIDE-IN extended permit udp host 12.1.1.2 host 11.1.1.1 eq 500
access-list OUTSIDE-IN extended permit udp host 12.1.1.2 host 11.1.1.1 eq 4500
access-group OUTSIDE-IN in interface outside
